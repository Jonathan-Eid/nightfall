cmake_minimum_required(VERSION 3.21)
project(meteor_ovr_bridge LANGUAGES C CXX)
add_subdirectory(../qsv qsv)
add_library(meteor-ovr-bridge SHARED bridge.cpp ../steamvr-driver/transport.c)
target_compile_features(meteor-ovr-bridge PRIVATE cxx_std_17 c_std_11)
target_compile_definitions(meteor-ovr-bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
find_package(Vulkan REQUIRED)
target_link_libraries(meteor-ovr-bridge PRIVATE meteor-qsv d3d11 dxgi ws2_32 ole32 uuid ksuser Vulkan::Vulkan)
add_executable(meteor-loopback-audio-probe ../windows_loopback_audio_probe.cpp)
target_compile_features(meteor-loopback-audio-probe PRIVATE cxx_std_17)
target_compile_definitions(meteor-loopback-audio-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(meteor-loopback-audio-probe PRIVATE ole32 uuid ksuser)
add_custom_command(TARGET meteor-ovr-bridge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:meteor-qsv> $<TARGET_FILE_DIR:meteor-ovr-bridge>)
add_library(meteor-windows-tracking SHARED ../meteor_tracking_client_bridge.cpp)
target_compile_features(meteor-windows-tracking PRIVATE cxx_std_17)
target_include_directories(meteor-windows-tracking PRIVATE ../../extensions/nightfall-xr/include)
set_target_properties(meteor-windows-tracking PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_link_libraries(meteor-windows-tracking PRIVATE ws2_32)
add_executable(meteor-ovr-service service.cpp)
target_compile_features(meteor-ovr-service PRIVATE cxx_std_17)
