cmake_minimum_required(VERSION 3.21)
project(meteor_qsv LANGUAGES CXX)
set(ONEVPL_INCLUDE "" CACHE PATH "Pinned Intel libvpl api directory")
get_filename_component(ONEVPL_SOURCE "${ONEVPL_INCLUDE}" DIRECTORY)
if(NOT EXISTS "${ONEVPL_SOURCE}/libvpl/CMakeLists.txt")
    include(FetchContent)
    FetchContent_Declare(meteor_onevpl
        GIT_REPOSITORY https://github.com/intel/libvpl.git
        GIT_TAG d77f9195cf495b937631607333288fd917ae8939)
    FetchContent_GetProperties(meteor_onevpl)
    if(NOT meteor_onevpl_POPULATED)
        FetchContent_Populate(meteor_onevpl)
    endif()
    set(ONEVPL_INCLUDE "${meteor_onevpl_SOURCE_DIR}/api")
    set(ONEVPL_SOURCE "${meteor_onevpl_SOURCE_DIR}")
endif()
# Include the MIT dispatcher in our DLL, avoiding a separate system-wide
# libvpl.dll installation. The Intel hardware runtime still comes from its driver.
set(BUILD_SHARED_LIBS OFF)
set(BUILD_TESTS OFF)
set(BUILD_EXAMPLES OFF)
set(INSTALL_DEV OFF)
set(INSTALL_EXAMPLES OFF)
add_subdirectory("${ONEVPL_SOURCE}" dispatcher EXCLUDE_FROM_ALL)
add_library(meteor-qsv SHARED qsv_encoder.cpp)
target_compile_definitions(meteor-qsv PRIVATE METEOR_QSV_STATIC_DISPATCHER)
target_compile_features(meteor-qsv PRIVATE cxx_std_17)
target_include_directories(meteor-qsv PRIVATE "${ONEVPL_INCLUDE}")
target_link_libraries(meteor-qsv PRIVATE VPL d3d11 dxgi)
add_executable(meteor-qsv-surface-probe ../windows_qsv_surface_probe.cpp)
target_compile_features(meteor-qsv-surface-probe PRIVATE cxx_std_17)
target_compile_definitions(meteor-qsv-surface-probe PRIVATE METEOR_QSV_STATIC_DISPATCHER)
target_include_directories(meteor-qsv-surface-probe PRIVATE "${ONEVPL_INCLUDE}")
target_link_libraries(meteor-qsv-surface-probe PRIVATE meteor-qsv VPL d3d11 dxgi)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(meteor-qsv-encode-test ../windows_qsv_encode_test.cpp)
    target_compile_features(meteor-qsv-encode-test PRIVATE cxx_std_17)
    target_link_libraries(meteor-qsv-encode-test PRIVATE meteor-qsv d3d11 dxgi)
endif()
